준비하고, 무작위로 나눈 자료로 학습 · 검증

import pandas as pd; from sklearn.model_selection import train_test_split; from sklearn.neighbors import KNeighborsClassifier; df = pd.read_csv('drive_log.csv'); cols = ['line_near', 'line_far', 'bright']
train, val = train_test_split(df, test_size=0.2, random_state=42); m1 = KNeighborsClassifier(n_neighbors=3).fit(train[cols], train['label']); print(round(m1.score(val[cols], val['label']), 3))
0.959
